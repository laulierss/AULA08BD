-- TABELA DE ESTOQUE DE VEÍCULOS
CREATE TABLE IF NOT EXISTS veiculos_estoque (
  id SERIAL PRIMARY KEY,
  modelo VARCHAR(80) NOT NULL,
  marca VARCHAR(40) NOT NULL,
  ano_fabricacao INT NOT NULL,
  preco DECIMAL(10,2) NOT NULL,
  quilometragem INT NOT NULL,
  combustivel VARCHAR(30) NOT NULL
);

-- -- MISSÃO 1
SELECT 
    modelo AS "Carro",
    marca AS "Fabricante",
    preco AS "Valor de Venda"
FROM veiculos_estoque;

-- -- MISSÃO 2
SELECT 
    preco,
    (preco * 1.10) AS "IPVA Estimado (4%)"
FROM veiculos_estoque;

-- -- MISSÃO 3
SELECT ano_fabricacao, quilometragem
FROM veiculos_estoque
WHERE (ano_fabricacao >= 2022) AND (quilometragem <= 30.000);

-- MISSÃO 4
SELECT modelo, preco
FROM veiculos_estoque
ORDER BY preco DESC LIMIT 3;
